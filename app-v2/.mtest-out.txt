  function makeClamps(count) {

    }

  }

  // 应用 K+1 个圆柱的新长度；两端都锁定的圆柱保持定点，不随自动模式改变
  function endInnerLocked(i) {
    if (i === 0) return !!state.clamps[0].r;
    return !!state.clamps[cylCount() - 1].l;
  }
  // desiredEndDelta：两端总长度要变化多少；prefer：优先吸收的一端；excluded：不变的一端
  function allocateEnds(desiredEndDelta, prefer, excluded, baseLengths) {
    var last = cylCount() - 1;
    var base = baseLengths || state.lengths;
    var e0 = base[0];
    var eL = base[last];
    var used = 0;
    function add(i, amount) {
      if (amount === 0) return;
      if (i === excluded) return;
      if (endInnerLocked(i)) return;
      var cur = i === 0 ? e0 : eL;
      var next = clamp(cur + amount, 30, 120);
      if (i === 0) e0 = next; else eL = next;
      used += next - cur;
    }
    if (prefer === 0 || prefer === last) {
      add(prefer, desiredEndDelta);
      add(prefer === 0 ? last : 0, desiredEndDelta - used);
    } else {
      add(0, desiredEndDelta / 2);
      add(last, desiredEndDelta / 2);
      add(0, desiredEndDelta - used);
      add(last, desiredEndDelta - used);
    }
    var achieved = (e0 - base[0]) + (eL - base[last]);
    if (Math.abs(achieved - desiredEndDelta) &gt; 0.01) return { ok: false };
    return { ok: true, e0: e0, eL: eL };
  }

  // 内部目标变化 Δ 全部由两个端部内端吸收（各 -Δ/2，一端到限位另一端补）
  function applyInternalTargets(newInternal, prefer) {
    var oldInternal = state.lengths.slice(1, -1);
    var delta = sum(newInternal) - sum(oldInternal);
    var alloc = allocateEnds(-delta, prefer, null);
    if (!alloc.ok) {
      showToast('端部行程不够，先减件或缩短目标');
      log('端部行程不够 · 操作未生效', 'warn');

    if (toastEl) { toastEl.classList.remove('show'); toastEl.textContent = ''; clearTimeout(toastTimer); }
    state.lengths = [alloc.e0].concat(newInternal).concat([alloc.eL]);
    var total = chainTotal(state.lengths);
    state.overLimit = total &gt; availableMm() + 0.001;
    state.requestedTotal = total;
    state.remaining = Math.max(0, availableMm() - total);
    render();
    updateAlerts();
    return true;
  }

      }

  }

  /* =========================================================================
     6. 分配算法
     ========================================================================= */
  function distribute(weights, total, min, max) {
    var n = weights.length;
    var result = new Array(n).fill(null);
    var active = weights.map(function (_, i) { return i; });
    var remaining = total;
    for (var guard = 0; guard &lt; 16 &amp;&amp; active.length; guard++) {
      var sumW = active.reduce(function (s, i) { return s + weights[i]; }, 0);
      var changed = false;
      for (var k = 0; k &lt; active.length; k++) {
        var idx = active[k];
        var v = remaining * (weights[idx] / sumW);
        if (v &lt; min - 1e-9 || v &gt; max + 1e-9) {
          result[idx] = clamp(v, min, max);
          remaining -= result[idx];
          active = active.filter(function (x) { return x !== idx; });
          changed = true;
          break;
        }
      }
      if (!changed) {
        active.forEach(function (i) { result[i] = remaining * (weights[i] / sumW); });
        active = [];
        break;
      }
    }
    if (active.length) {
      var rest = remaining / active.length;
      active.forEach(function (i) { result[i] = clamp(rest, min, max); });
    }
    return result.map(function (v) { return v == null ? min : v; });
  }

      weights.push((humidityWeight(state.hangers[i].humidity) +
        humidityWeight(state.hangers[i + 1].humidity)) / 2);
    }
    var budget = Math.max(targetMm * free.length, CYL_MIN * free.length);
    var lens = distribute(weights, budget, CYL_MIN, CYL_MAX);
    if (applyInternalTargets(lens, null)) {
      log('按湿度加权 · 目标间距 ' + fmt(state.targetCm) + ' cm', 'info');
    }
  }

    }
  }

    for (var cm = 0; cm &lt;= rodCm + 1e-6; cm += 10) {
      var tick = document.createElement('i');
      tick.className = 'tick' + (Math.abs(cm % labelStep) &lt; 1e-6 ? ' major' : '');
      tick.style.left = (cm / rodCm * 100) + '%';
      frag.appendChild(tick);
    }
    labelsLoop();
    function labelsLoop() {
      var labels = [];
      for (var l = 0; l &lt; rodCm - 1e-6; l += labelStep) labels.push(l);
      labels.push(rodCm);
      labels.forEach(function (value) {
        var label = document.createElement('span');
        label.className = 'ruler-label' + (value === 0 ? ' edge-l' : Math.abs(value - rodCm) &lt; 1e-6 ? ' edge-r' : '');
        label.style.left = (value / rodCm * 100) + '%';
        label.textContent = Math.round(value);
        frag.appendChild(label);
      });
    }
    ruler.appendChild(frag);
  }

  /* ---------------- 物理拖动：端点锁 + 传播 ----------------
     只移动被拖端点；接缝和相邻自由端跟随；
     碰到锁死端点立即拒绝；太长/太短也拒绝。 */
  function endLocked(i, side) {
    var c = state.clamps[i];
    return side === 'l' ? c.l : c.r;
  }
  function lenInRange(i, value) {
    return value &gt;= lenMinFor(i) - 0.01 &amp;&amp; value &lt;= lenMaxFor(i) + 0.01;
  }
  function blockedMsg(j, side, target) {
    var sw = side === 'l' ? '左' : '右';
    return '圆柱 ' + pad2(j + 1) + ' ' + sw + '端固定挡住了，先松 ' + pad2(j + 1) +
      ' ' + sw + '端，或缩短 ' + pad2(target + 1);
  }
  function tooLongMsg(i) {
    return '圆柱 ' + pad2(i + 1) + ' 行程到头了，先缩短 ' + pad2(i + 1);
  }

  // 返回 {lengths} 或 {error}
  function computeDrag(startLengths, index, side, deltaMm) {
    var arr = startLengths.slice();
    var c = state.clamps;
    if (isEndCylinder(index)) return { error: '端部固定，不可调' };
    if (side === 'l') {
      if (c[index].l) return { error: '这端已锁定' };
      if (!c[index].r) return { error: '先锁一端' };
      arr[index] -= deltaMm;
      if (!lenInRange(index, arr[index])) return { error: tooLongMsg(index) };
      var shift = deltaMm;
      var j = index - 1;
      while (j &gt;= 0) {
        if (c[j].r) return { error: blockedMsg(j, 'r', index) };
        if (c[j].l) {
          arr[j] += shift;
          if (!lenInRange(j, arr[j])) return { error: tooLongMsg(j) };
          return { lengths: arr };
        }
        // 两端都松：整段平移，长度不变，继续往左传播
        j--;
      }
      return { error: '左侧固定端挡住了，先松固定端或缩短 ' + pad2(index + 1) };
    }
    if (c[index].r) return { error: '这端已锁定' };
    if (!c[index].l) return { error: '先锁一端' };
    arr[index] += deltaMm;
    if (!lenInRange(index, arr[index])) return { error: tooLongMsg(index) };
    var shift2 = deltaMm;
    var k = index + 1;
    while (k &lt; cylCount()) {
      if (c[k].l) return { error: blockedMsg(k, 'l', index) };
      if (c[k].r) {
        arr[k] -= shift2;
        if (!lenInRange(k, arr[k])) return { error: tooLongMsg(k) };
        return { lengths: arr };
      }
      // 两端都松：整段平移，长度不变，继续往右传播
      k++;
    }
    return { error: '右侧固定端挡住了，先松固定端或缩短 ' + pad2(index + 1) };
  }

  function startHandleDrag(ev, index, side, el) {
    if (ev.button !== undefined &amp;&amp; ev.button !== 0) return;
    ev.preventDefault();
    ev.stopPropagation();
    var endCyl = isEndCylinder(index);
    if (endCyl) {
      var innerSide = index === 0 ? 'r' : 'l';
      if (side !== innerSide) { showToast('外端锚定，不可调'); return; }
      if (endLocked(index, side)) { showToast('内端已锁定'); return; }
    } else {
      if (endLocked(index, side)) { showToast('这端已锁定'); return; }
      var c0 = state.clamps[index];
      if (!c0.l &amp;&amp; !c0.r) { showToast('先锁一端'); return; }
    }
    clearTimeout(applyTimer);
    stopMotion();
    state.mode = 'manual';
    updateModeUI();
    dragState = {
      kind: 'handle',
      index: index,
      side: side,
      el: el,
      startX: ev.clientX,
      startLengths: state.lengths.slice(),
      scale: view.scale || 0.2,
      pointerId: ev.pointerId,
      lastMsg: ''
    };
    el.classList.add('dragging');
    try { el.setPointerCapture(ev.pointerId); } catch (err) {}
  }

  function onHandleMove(ev) {
    if (!dragState || dragState.kind !== 'handle') return;
    if (dragState.pointerId !== undefined &amp;&amp; ev.pointerId !== dragState.pointerId) return;
    var delta = (ev.clientX - dragState.startX) / dragState.scale;
    var idx = dragState.index, side = dragState.side, last = cylCount() - 1;
    var candidate = dragState.startLengths.slice();
    var old = candidate[idx];
    var next = old + (side === 'r' ? delta : -delta);
    var lo = isEndCylinder(idx) ? 30 : CYL_MIN;
    var hi = isEndCylinder(idx) ? 120 : CYL_MAX;
    if (next &lt; lo - 0.01 || next &gt; hi + 0.01) {
      if (dragState.lastMsg !== '行程到头了') {
        dragState.lastMsg = '行程到头了';
        showToast('圆柱 ' + pad2(idx + 1) + ' 行程到头了');
      }
      return;
    }
    candidate[idx] = next;
    var change = next - old;
    var alloc;
    if (isEndCylinder(idx)) {
      var other = idx === 0 ? last : 0;
      alloc = allocateEnds(-change, other, idx, dragState.startLengths);
      if (!alloc.ok) {
        if (dragState.lastMsg !== '端部行程不够') {
          dragState.lastMsg = '端部行程不够';
          showToast('端部行程不够，先松另一端或缩短目标');
        }
        return;
      }
      candidate[0] = alloc.e0;
      candidate[last] = alloc.eL;
    } else {
      var prefer = side === 'r' ? last : 0;
      alloc = allocateEnds(-change, prefer, null, dragState.startLengths);
      if (!alloc.ok) {
        if (dragState.lastMsg !== '端部行程不够') {
          dragState.lastMsg = '端部行程不够';
          showToast('端部行程不够，先缩短目标');
        }
        return;
      }
      candidate[0] = alloc.e0;
      candidate[last] = alloc.eL;
    }
    state.lengths = candidate;
    var total = chainTotal(state.lengths);
    state.overLimit = total &gt; availableMm() + 0.001;
    state.requestedTotal = total;
    state.remaining = Math.max(0, availableMm() - total);
    render();
  }

  function endHandleDrag(ev) {
    if (!dragState || dragState.kind !== 'handle') return;
    var d = dragState;
    dragState = null;
    if (d.el) {
      d.el.classList.remove('dragging');
      try { d.el.releasePointerCapture(d.pointerId); } catch (err) {}
    }
    log('拖动 · 圆柱 ' + pad2(d.index + 1) + ' ' + (d.side === 'l' ? '左' : '右') +
      '端 · 总长 ' + Math.round(chainTotal(state.lengths)) + ' mm', 'gold');
    render();
    updateAlerts();
  }

        else if (state.clamps[index].l &amp;&amp; state.clamps[index].r) showToast('两端锁定，先松一端');

  function toggleClamp(index, side) {
    if (isEndCylinder(index)) {
      var outerSide = index === 0 ? 'l' : 'r';
      if (side === outerSide) {
        showToast('外端锚定，不可调');
        return;
      }
    }
    var c = state.clamps[index];
    if (!c) return;
    c[side] = !c[side];
    log('圆柱 ' + pad2(index + 1) + ' ' + (side === 'l' ? '左' : '右') + '端夹头 · ' +
      (c[side] ? '锁定' : '松开'), 'gold');
    render();
  }

  /* 按住圆柱左右拖 = 改该段长度；两端都锁拒绝；拖动时暂停自动模式 */
  function startDrag(ev, index, el) {
    if (ev.button !== undefined &amp;&amp; ev.button !== 0) return;
    if (ev.target &amp;&amp; ev.target.closest &amp;&amp; ev.target.closest('.clamp')) return;
    if (isEndCylinder(index)) {
      showToast('端部固定，先解锁再调');
      log('圆柱 ' + pad2(index + 1) + ' 是固定端 · 端部固定，先解锁再调', 'warn');
      return;
    }
    var c = state.clamps[index];
    if (c &amp;&amp; c.l &amp;&amp; c.r) {
      showToast('先松一端');
      log('圆柱 ' + pad2(index + 1) + ' 两端夹头都锁住 · 先松一端再拖', 'warn');
      return;
    }
    ev.preventDefault();
    clearTimeout(applyTimer);
    stopMotion();
    state.mode = 'manual';
    updateModeUI();
    dragState = {
      index: index,
      el: el,
      startX: ev.clientX,
      startLen: state.lengths[index],
      scale: view.scale || 0.2,
      pointerId: ev.pointerId
    };
    el.classList.add('dragging');
    try { el.setPointerCapture(ev.pointerId); } catch (err) {}
  }

  function onDragMove(ev) {
    if (!dragState || (dragState.pointerId !== undefined &amp;&amp; ev.pointerId !== dragState.pointerId)) return;
    var dx = ev.clientX - dragState.startX;
    var deltaMm = dx / dragState.scale;
    var lo = lenMinFor(dragState.index);
    var hi = lenMaxFor(dragState.index);
    var next = clamp(dragState.startLen + deltaMm, lo, hi);
    state.lengths[dragState.index] = next;
    var total = chainTotal(state.lengths);
    state.overLimit = total &gt; availableMm() + 0.001;
    state.requestedTotal = total;
    state.remaining = Math.max(0, availableMm() - total);

    }
    log('拖动 · 圆柱 ' + pad2(d.index + 1) + ' = ' + (state.lengths[d.index] / 10).toFixed(1) +
      ' cm · 总长 ' + Math.round(chainTotal(state.lengths)) + ' mm', 'gold');

    updateAlerts();

  function setRod(mm) {
    var next = clamp(Number(mm), ROD_MIN_MM, ROD_MAX_MM);
    if (Math.abs(next - state.rodMm) &lt; 0.001) return;
    state.rodMm = next;
    var fit = fitLengths(state.lengths);
    state.lengths = fit.lengths.slice();
    state.overLimit = fit.compressed;
    state.requestedTotal = fit.requestedTotal;
    state.remaining = fit.remaining;
    buildRuler();
    updateRodUI();
    measure();
    updateAlerts();
  }

  function changeK(delta) {
    if (delta &gt; 0) {
      if (K() &gt;= K_MAX) return;
      var oldK = K();
      var p = HANGER_PROFILES[oldK];
      state.hangers.push({ humidity: p.humidity, battery: p.battery });
      // 在最后一个端部夹头之前插入一个内部圆柱 + 接缝
      state.lengths.splice(state.lengths.length - 1, 0, clamp(state.targetCm * 10, CYL_MIN, CYL_MAX));
      state.clamps.splice(oldK, 0, { l: false, r: false });
      var fit = fitLengths(state.lengths);
      state.lengths = fit.lengths.slice();
      state.overLimit = fit.compressed;
      state.requestedTotal = fit.requestedTotal;
      state.remaining = fit.remaining;
      log('加一件 · 接缝 ' + pad2(K()) + ' + 伸缩圆柱 ' + pad2(K()) +
        '（圆柱 ' + cylCount() + ' 个，两端夹头保留）', 'gold');
      syncDom();
      var enter = cylEls[oldK];
      if (enter) {
        enter.classList.add('cyl-enter');
        requestAnimationFrame(function () {
          requestAnimationFrame(function () { enter.classList.remove('cyl-enter'); });
        });
      }
    } else {
      if (K() &lt;= K_MIN) return;
      var oldK2 = K();
      state.hangers.pop();
      state.lengths.splice(state.lengths.length - 2, 1);
      state.clamps.splice(oldK2 - 1, 1);
      var fit2 = fitLengths(state.lengths);
      state.lengths = fit2.lengths.slice();
      state.overLimit = fit2.compressed;
      state.requestedTotal = fit2.requestedTotal;
      state.remaining = fit2.remaining;
      log('减一件 · 移除接缝 ' + pad2(oldK2) + ' + 伸缩圆柱 ' + pad2(oldK2), 'info');
      syncDom();
    }
    updateKUI();
    updateStats();
    if (state.mode === 'equal' || state.mode === 'weighted' || state.mode === 'collect') {
      applyMode();
    } else {
      render();
    }
    updateAlerts();
  }
