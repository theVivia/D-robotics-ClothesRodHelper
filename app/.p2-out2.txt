    var b = h &lt;= 50 ? stops[1] : stops[2];

      frag.appendChild(tick);

      label.style.left = (cm / ROD_CM * 100) + '%';
      label.textContent = cm;
      ruler.appendChild(label);
    });
  }

  function render() {

        ? 'linear-gradient(90deg, rgba(240,122,99,.5), var(--coral))'
        : '';
    });

    // ---- 间距标注 / 气流 / 读数 ----
    for (var i = 0; i &lt; order.length - 1; i++) {
      var a = order[i];
      var b = order[i + 1];
      var clear = (b.pos - a.pos) - (a.bulk + b.bulk) / 2;
      var displayClear = Math.max(0, clear);
      var left = centerX(a.pos) + garmentPx(a.bulk) / 2;
      var right = centerX(b.pos) - garmentPx(b.bulk) / 2;
      var width = Math.max(0, right - left);
      var tone = clearTone(clear);

      var dim = dimEls[i];
      dim.el.style.left = left + 'px';
      dim.el.style.width = width + 'px';
      dim.el.dataset.tone = tone;
      dim.label.style.visibility = width &gt;= (view.w &lt; 520 ? 30 : 26) ? 'visible' : 'hidden';
      dim.value.textContent = fmt(displayClear);
      dim.el.style.opacity = width &lt; 3 ? '0' : '1';

      var flow = flowEls[i];
      flow.style.left = left + 'px';
      flow.style.width = width + 'px';
      flow.style.opacity = width &gt; 12 ? '' : '0';

      chan.pair.textContent = pad2(a.id) + '—' + pad2(b.id);
      chan.val.textContent = fmt(displayClear);
      chan.sub.textContent = '中心距 ' + fmt(b.pos - a.pos) + 'cm';
      chan.el.dataset.tone = tone;
    }

    // ---- 顶部统计 ----
    var clears = [];
    for (var j = 0; j &lt; order.length - 1; j++) {
      clears.push(Math.max(0, (order[j + 1].pos - order[j].pos) - (order[j].bulk + order[j + 1].bulk) / 2));
    }
    var avg = clears.reduce(function (s, c) { return s + c; }, 0) / clears.length;
    var min = Math.min.apply(null, clears);
    var hits = clears.filter(function (c) { return c &gt;= state.targetGap - CLEAR_TOL; }).length;

    statAvg.innerHTML = fmt(avg) + '&lt;em&gt;cm&lt;/em&gt;';
    statMin.innerHTML = fmt(min) + '&lt;em&gt;cm&lt;/em&gt;';
    statMin.className = 'stat-value' +
      (min &lt; MIN_CLEAR ? ' warn' : min &lt; state.targetGap - CLEAR_TOL ? ' gold' : '');
    statTarget.innerHTML = fmt(state.targetGap) + '&lt;em&gt;cm&lt;/em&gt;';
    statOk.textContent = hits + '/' + clears.length;
    statOk.className = 'stat-value ' + (hits === clears.length ? 'ok' : 'warn');
  }

  /* =========================================================================
     7. 几何求解：目标位置
     ========================================================================= */
  // 按顺序 + 各段净风道，计算每个单元的目标中心位置（整体居中，越界自动缩放）
  function layoutTargets(order, gaps, center) {
    var bulkSum = 0;
    order.forEach(function (it) { bulkSum += it.bulk; });
    var avail = ROD_CM - 2 * EDGE_CM;
    var gapSum = gaps.reduce(function (s, g) { return s + g; }, 0);
    var gs = gaps.slice();

    if (bulkSum + gapSum &gt; avail) {
      var k = (avail - bulkSum) / gapSum;
      gs = gs.map(function (g) { return Math.max(MIN_CLEAR, g * k); });
      gapSum = gs.reduce(function (s, g) { return s + g; }, 0);
    }

    var span = bulkSum + gapSum;
    var cursor = clamp(center - span / 2, EDGE_CM, ROD_CM - EDGE_CM - span);
    var targets = new Map();
    order.forEach(function (it, i) {
      if (i &gt; 0) cursor += gs[i - 1];
      targets.set(it.id, cursor + it.bulk / 2);
      cursor += it.bulk;
    });
    return targets;
  }

  // 智能分配基准：环境越湿，基准间隙越大（0.80—1.70 倍）
  function smartBaseGap() {
    var t = (clamp(state.env, ENV_MIN, ENV_MAX) - ENV_MIN) / (ENV_MAX - ENV_MIN);
    var factor = 0.8 + t * 0.9;
    return clamp(state.targetGap * factor, MIN_CLEAR, MAX_CLEAR);
  }

  // 带上下限的按权重分配（water-filling）：返回各项间隙，总和 = total
  function distribute(weights, total, min, max) {
    var n = weights.length;
    var result = new Array(n).fill(null);
    var active = weights.map(function (_, i) { return i; });
    var remaining = total;

    for (var guard = 0; guard &lt; 16 &amp;&amp; active.length; guard++) {
      var sumW = active.reduce(function (s, i) { return s + weights[i]; }, 0);
      var changed = false;
      for (var k = 0; k &lt; active.length; k++) {
        var idx = active[k];
        var v = remaining * (weights[idx] / sumW);
        if (v &lt; min - 1e-9 || v &gt; max + 1e-9) {
          result[idx] = clamp(v, min, max);
          remaining -= result[idx];
          active = active.filter(function (x) { return x !== idx; });
          changed = true;
          break;
        }
      }
      if (!changed) {
        for (var m = 0; m &lt; active.length; m++) {
          result[active[m]] = remaining * (weights[active[m]] / sumW);
        }
        active = [];
        break;
      }
    }

    if (active.length) {
      var rest = remaining / active.length;
      active.forEach(function (i) { result[i] = clamp(rest, min, max); });
    }
    return result.map(function (v) { return v == null ? min : v; });
  }

  /* =========================================================================
     8. 运动：所有位移走同一条平滑动画管线（真机可替换为 MQTT 目标下发）
     ========================================================================= */
  var raf = window.requestAnimationFrame || function (fn) { return setTimeout(function () { fn(Date.now()); }, 16); };

  function stopMotion() {
    state.motion = null;
    state.arcIds = [];
    items.forEach(function (it) { it.lift = 0; });
  }

  function startMotion(targets, opts) {
    opts = opts || {};
    stopMotion();
    var from = new Map();
    var maxDelta = 0;
    items.forEach(function (it) {
      from.set(it.id, it.pos);
      var to = targets.get(it.id);
      if (to != null) maxDelta = Math.max(maxDelta, Math.abs(to - it.pos));
    });
    var baseDur = opts.duration || 900;
    var dur = REDUCED ? 260 : clamp(baseDur + maxDelta * 8, baseDur, baseDur + 500);
    var arcs = opts.arcIds || (opts.arcId ? [opts.arcId] : []);

    state.motion = {
      from: from,
      to: targets,
      t0: performance.now(),
      dur: dur,
      arcIds: arcs,
      onDone: opts.onDone || null
    };
    state.arcIds = arcs;
    state.running = true;
    updateStatus();
    raf(motionTick);
  }

  function motionTick(now) {
    var m = state.motion;
    if (!m) return;
    var t = clamp((now - m.t0) / m.dur, 0, 1);
    var e = easeInOutCubic(t);
    var liftPx = view.w &lt; 520 ? 26 : 38;

    items.forEach(function (it) {
      var to = m.to.get(it.id);
      if (to == null) return;
      it.pos = lerp(m.from.get(it.id), to, e);
      it.lift = m.arcIds.indexOf(it.id) !== -1 ? Math.sin(Math.PI * t) * liftPx : 0;
    });
    render();

    if (t &gt;= 1) {
      state.motion = null;
      state.arcIds = [];
      items.forEach(function (it) { it.lift = 0; });
      render();
      updateStatus();
      if (m.onDone) m.onDone();
    } else {
      raf(motionTick);
    }
  }

  function runEqual() {
    var order = byPos();
    var gap = clamp(state.targetGap, MIN_CLEAR, MAX_CLEAR);
    var gaps = order.slice(0, -1).map(function () { return gap; });
    var targets = layoutTargets(order, gaps, ROD_CM / 2);
    log('一键等距 · 目标净风道 ' + fmt(gap) + ' cm', 'info');
    startMotion(targets, {
      duration: 900,
      onDone: function () { log('等距完成 · 三段风道已对齐', 'ok'); }
    });
  }

  function runCollect() {
    var order = byPos();
    var targets = new Map();
    var cursor = ROD_CM - EDGE_CM;
    for (var i = order.length - 1; i &gt;= 0; i--) {
      var it = order[i];
      var c = cursor - it.bulk / 2;
      targets.set(it.id, c);
      cursor = c - it.bulk / 2 - MIN_CLEAR;
    }
    log('收衣归拢 · 单元向右端收纳位靠拢，保持 ' + MIN_CLEAR + ' cm 安全间隙', 'info');
    startMotion(targets, {
      duration: 1200,
      onDone: function () { log('收衣完成 · 可摘下衣物', 'ok'); }
    });
  }

  // 顺序冲突：两件最湿的衣物被别的衣物隔开（无法共用一条大风道）
  function findConflict(order) {
    var wet = order.filter(function (it) { return it.humidity &gt;= 65; })
      .sort(function (a, b) { return b.humidity - a.humidity; });
    if (wet.length &lt; 2) return null;
    var a = wet[0], b = wet[1];
    var ia = order.indexOf(a), ib = order.indexOf(b);
    if (Math.abs(ia - ib) === 1) return null; // 已相邻，风道已经连在一起
    var between = order.slice(Math.min(ia, ib) + 1, Math.max(ia, ib));
    if (!between.length) return null;
    var driest = between.reduce(function (x, y) { return y.humidity &lt; x.humidity ? y : x; });
    return { wetA: a, wetB: b, between: between, driest: driest };
  }

  function runSmart() {
    var order = byPos();
    var newOrder = order.slice();
    var arcIds = [];
    var base = smartBaseGap();
    var conflict = findConflict(order);

    // ---- 顺序冲突：默认不换序（与 P0/P1 真机一致）；只有打开 P2 概念才演示抬升越障 ----
    if (conflict) {
      var betweenText = conflict.between.map(tag).join(' / ');
      var conflictText = '顺序冲突 · 最湿的 ' + tag(conflict.wetA) + ' / ' + tag(conflict.wetB) +
        ' 被 ' + betweenText + ' 隔开';
      if (!state.p2Lift) {
        log(conflictText + ' · P0/P1 真机不能换序，按当前顺序分配', 'warn');
      } else {
        log(conflictText + ' · P2 概念演示允许抬升越障', 'warn');
        // P2 概念：把两件最湿衣物之间的&quot;挡路单元&quot;整体移到最远端（保持它们彼此先后）
        newOrder = newOrder.filter(function (it) { return conflict.between.indexOf(it) === -1; });
        newOrder = newOrder.concat(conflict.between);
        arcIds = conflict.between.map(function (it) { return it.id; });
        log('避让重排 · 挡路的低湿单元 ' + betweenText + ' 移至最远端，让高湿区相邻', 'gold');
        showToast('P2 概念 · 抬升越障重排');
      }
    }

    // ---- 按湿度权重分隙：湿的相邻对，间隙更大 ----
    var weights = [];
    for (var j = 0; j &lt; newOrder.length - 1; j++) {
      weights.push((humidityWeight(newOrder[j]) + humidityWeight(newOrder[j + 1])) / 2);
    }
    var budget = base * (newOrder.length - 1);
    var gaps = distribute(weights, budget, MIN_CLEAR, MAX_CLEAR);
    var targets = layoutTargets(newOrder, gaps, ROD_CM / 2);

    log('智能分配 · 环境 ' + state.env + '% RH → 基准净风道 ' + fmt(base) + ' cm' +
      (state.p2Lift ? ' · P2 概念开启' : ' · 保持当前顺序'), 'info');

    var finalOrder = newOrder;
    startMotion(targets, {
      duration: arcIds.length ? 1400 : 1050,
      arcIds: arcIds,
      onDone: function () {
        var show = [];
        for (var k = 0; k &lt; finalOrder.length - 1; k++) {
          show.push(fmt(gaps[k]));
        }
        log('分配完成 · 风道 ' + show.join(' / ') + ' cm', 'ok');
      }
    });
  }

  function applyMode() {
    if (state.estop) return;
    state.running = true;
    if (state.mode === 'manual') {
      log('手动模式 · 拖动任意模块调整位置', 'info');
      render();
      updateStatus();
      return;
    }
    if (state.mode === 'equal') runEqual();
    else if (state.mode === 'smart') runSmart();
    else if (state.mode === 'collect') runCollect();
    updateStatus();
  }

  /* =========================================================================
     10. 拖动：限制在相邻单元之间，不互相穿过
     ========================================================================= */
  function clampDrag(it, desired) {
    var order = byPos();
    var idx = order.indexOf(it);
    var half = it.bulk / 2;
    var lo = EDGE_CM + half;
    var hi = ROD_CM - EDGE_CM - half;

    if (idx &gt; 0) {
      var prev = order[idx - 1];
      lo = Math.max(lo, prev.pos + prev.bulk / 2 + half + MIN_CLEAR);
    }
    if (idx &lt; order.length - 1) {
      var next = order[idx + 1];
      hi = Math.min(hi, next.pos - next.bulk / 2 - half - MIN_CLEAR);
    }
    it.pos = clamp(desired, lo, Math.max(lo, hi));
  }

  function bindDrag() {
    items.forEach(function (it) {
      var root = els.get(it.id).root;

      root.addEventListener('pointerdown', function (ev) {
        if (ev.button !== 0 &amp;&amp; ev.pointerType === 'mouse') return;
        if (state.estop) {
          log('急停锁定中 · 请先点击「复位并启动」', 'warn');
          shake(btnStart);
          return;
        }
        ev.preventDefault();
        stopMotion();
        if (state.mode !== 'manual') {
          state.mode = 'manual';
          updateModeUI();
          log('切换到手动模式 · 拖动模块调整位置', 'info');
        }
        state.running = true;
        state.drag = {
          id: it.id,
          pointerId: ev.pointerId,
          startX: ev.clientX,
          startPos: it.pos,
          moved: false
        };
        try { root.setPointerCapture(ev.pointerId); } catch (err) { /* 忽略 */ }
        root.classList.add('dragging');
        updateStatus();
      });

      root.addEventListener('pointermove', function (ev) {
        var d = state.drag;
        if (!d || d.id !== it.id || ev.pointerId !== d.pointerId) return;
        var dxCm = (ev.clientX - d.startX) / view.scale;
        if (Math.abs(dxCm) &gt; 0.15) d.moved = true;
        clampDrag(it, d.startPos + dxCm);
        render();
      });

      function endDrag() {
        var d = state.drag;
        if (!d || d.id !== it.id) return;
        state.drag = null;
        root.classList.remove('dragging');
        if (d.moved) {
          var order = byPos();
          var idx = order.indexOf(it);
          var parts = [];
          if (idx &gt; 0) {
            var pl = (it.pos - it.bulk / 2) - (order[idx - 1].pos + order[idx - 1].bulk / 2);
            parts.push('左 ' + fmt(pl));
          }
          if (idx &lt; order.length - 1) {
            var pr = (order[idx + 1].pos - order[idx + 1].bulk / 2) - (it.pos + it.bulk / 2);
            parts.push('右 ' + fmt(pr));
          }
          log('手动调整 · ' + tag(it) + ' 位置 ' + fmt(it.pos) + ' cm，净风道 ' + parts.join(' / ') + ' cm', 'ok');
        }
        render();
        updateStatus();
      }

      root.addEventListener('pointerup', endDrag);
      root.addEventListener('pointercancel', endDrag);
    });
  }

  function bindControls() {
    document.querySelectorAll('.mode').forEach(function (btn) {
      btn.addEventListener('click', function () {
        if (state.estop) {
          log('急停锁定中 · 请先点击「复位并启动」', 'warn');
          shake(btn);
          return;
        }
        var next = btn.dataset.mode;
        if (next === state.mode) {
          applyMode();          // 重复点击 = 再次执行当前模式
        } else {
          state.mode = next;
          updateModeUI();
          applyMode();
        }
        state.running = true;
        updateStatus();
      });
    });

    // 目标净风道
    gapRange.addEventListener('input', function () {
      state.targetGap = Number(gapRange.value);
      updateControlUI();
      if (state.running &amp;&amp; !state.estop &amp;&amp; (state.mode === 'equal' || state.mode === 'smart')) {
        scheduleApply();
      }
    });

    // 环境湿度：只影响智能分配的基准间隙
    envRange.addEventListener('input', function () {
      state.env = Number(envRange.value);
      updateControlUI();
      if (state.running &amp;&amp; !state.estop &amp;&amp; state.mode === 'smart') scheduleApply();
    });

    // P2 概念开关：默认关闭；开启后才演示&quot;抬升越障&quot;换序
    p2Toggle.addEventListener('change', function () {
      if (state.estop) {
        p2Toggle.checked = state.p2Lift;
        log('急停锁定中 · 无法切换 P2 概念演示', 'warn');
        return;
      }
      state.p2Lift = p2Toggle.checked;
      updateP2UI();
      log(state.p2Lift
        ? 'P2 概念演示开启 · 抬升越障（硬件需抬升机构，P0/P1 真机不具备）'
        : 'P2 概念演示关闭 · 智能分配恢复为保持当前顺序', state.p2Lift ? 'warn' : 'info');
      if (state.mode === 'smart' &amp;&amp; state.running) scheduleApply();
    });

    // 开始 / 执行（急停时变为复位）
    btnStart.addEventListener('click', function () {
      if (state.estop) {
        state.estop = false;
        stage.classList.remove('estop');
        log('急停复位 · 系统重新使能', 'ok');
        state.running = true;
        applyMode();
        updateStatus();
        return;
      }
      if (state.motion) {
        log('动作执行中 · 请稍候', 'info');
        return;
      }
      state.running = true;
      applyMode();
      updateStatus();
    });

    // 急停：立即中断动画、锁定拖动，必须手动复位
    btnEstop.addEventListener('click', function () {
      if (state.estop) return;
      state.estop = true;
      state.running = false;
      stopMotion();
      stage.classList.add('estop');
      log('⛔ 急停触发 · 全部模块立即锁定，需人工复位', 'err');
      render();
      updateStatus();
    });

    document.getElementById('clearLog').addEventListener('click', function () {
      logEl.innerHTML = '';
    });
  }

  /* =========================================================================
     13. 尺寸自适应 + 启动

  function measure() {
    var w = stage.clientWidth;
    if (!w) return;
    view.w = w;
    view.pad = w &lt; 520 ? 14 : 24;
    view.scale = (w - view.pad * 2) / ROD_CM;
    stage.style.setProperty('--rod-pad', view.pad + 'px');
    render();
  }

  function initLogs() {
    log('HangFlow 演示模式启动 · 前端模拟，未连接硬件', 'gold');
    log('MQTT 已连接（模拟）· RDK X5 中枢在线', 'ok');
    log('4 个套挂单元上线 · 电量正常', 'info');
    log('目标净风道 ' + fmt(state.targetGap) + ' cm · 安全下限 ' + MIN_CLEAR + ' cm（g_min）', 'info');

    // 用当前真实排布生成提示，保证日志与画面一致
    var order = byPos();
    var conflict = findConflict(order);
    if (conflict) {
      var betweenText = conflict.between.map(tag).join(' / ');
      log('检测到 ' + tag(conflict.wetA) + ' / ' + tag(conflict.wetB) + ' 高湿被 ' +
        betweenText + ' 隔开 · 默认保持顺序分配', 'warn');
    } else {
      var narrow = (order[1].pos - order[0].pos) - (order[0].bulk + order[1].bulk) / 2;
      if (narrow &lt; state.targetGap - CLEAR_TOL) {
        log('检测到 ' + tag(order[0]) + '–' + tag(order[1]) + ' 净风道 ' + fmt(narrow) +
          ' cm 未达目标 · 建议执行智能分配', 'warn');
      } else {
        log('手动模式 · 拖拽模块或点「智能分配」看风道打开', 'info');
      }
    }
  }

  // ---- 初始化 ----
  buildUnits();
  buildDims();
  buildFlows();
  buildChannels();
  buildRows();
  buildRuler();
  bindDrag();
  bindControls();
  updateModeUI();
  updateControlUI();
  updateP2UI();
  updateBatteryUI();
  measure();
  initLogs();
  updateStatus();

  if (window.ResizeObserver) {
    new ResizeObserver(measure).observe(stage);
  } else {
    window.addEventListener('resize', measure);
  }

  // 模拟 MQTT 往返时延
  setInterval(function () {
    latencyEl.textContent = (9 + Math.floor(Math.random() * 35)) + 'ms';
  }, 2800);

  // 模拟缓慢掉电
  setInterval(function () {
    if (state.estop) return;
    var it = items[Math.floor(Math.random() * items.length)];
    if (it.battery &gt; 18) {
      it.battery -= 1;
      updateBatteryUI();
    }
  }, 15000);

})();
